function projectOnSegment(pLat, pLng, aLat, aLng, bLat, bLng) {
-- Defines a function to find the closest point on a line segment between A and B from point P.
LINE 78:
  const dx = bLng - aLng, dy = bLat - aLat;
-- Calculates the delta X (longitude) and delta Y (latitude) of the segment AB.
LINE 79:
  const lenSq = dx * dx + dy * dy;
-- Calculates the squared length of the segment to avoid expensive square root operations.
LINE 80:
  if (lenSq === 0) return { lat: aLat, lng: aLng };
-- If the segment length is zero (A and B are the same), returns point A.
LINE 81:
  const t = Math.max(0, Math.min(1, ((pLng - aLng) * dx + (pLat - aLat) * dy) / lenSq));
-- Calculates the parameter 't' representing the normalized projection of point P onto the line, clamped between 0 and 1.
LINE 82:
  return { lat: aLat + t * dy, lng: aLng + t * dx };
-- Returns the interpolated latitude and longitude on the segment based on 't'.
LINE 83:
}

LINE 97:
